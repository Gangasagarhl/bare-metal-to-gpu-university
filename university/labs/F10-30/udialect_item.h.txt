struct UMissionItem {
    static constexpr std::uint32_t kId = 12;
    static constexpr std::uint8_t kCrcExtra = 58;
    static constexpr std::uint8_t kLen = 27;
    std::int32_t x{};
    std::int32_t y{};
    float z{};
    std::uint16_t seq{};
    std::uint16_t command{};
    std::uint8_t target_system{};
    std::uint8_t target_component{};
    std::uint8_t frame{};
    std::array<char, 8> label{};
    // write the payload in wire order; returns its full length
    std::size_t pack(std::uint8_t* p) const
    {
        std::memcpy(p + 0, &x, 4);
        std::memcpy(p + 4, &y, 4);
        std::memcpy(p + 8, &z, 4);
        std::memcpy(p + 12, &seq, 2);
        std::memcpy(p + 14, &command, 2);
        std::memcpy(p + 16, &target_system, 1);
        std::memcpy(p + 17, &target_component, 1);
        std::memcpy(p + 18, &frame, 1);
        std::memcpy(p + 19, &label, 8);
        return kLen;
    }
    // read a payload; bytes beyond n (truncated trailing zeros) count as zero
    static UMissionItem unpack(const std::uint8_t* p, std::size_t n)
    {
        std::uint8_t full[kLen] = {};
        std::memcpy(full, p, n < kLen ? n : kLen);
        UMissionItem m;
        std::memcpy(&m.x, full + 0, 4);
        std::memcpy(&m.y, full + 4, 4);
        std::memcpy(&m.z, full + 8, 4);
        std::memcpy(&m.seq, full + 12, 2);
        std::memcpy(&m.command, full + 14, 2);
        std::memcpy(&m.target_system, full + 16, 1);
        std::memcpy(&m.target_component, full + 17, 1);
        std::memcpy(&m.frame, full + 18, 1);
        std::memcpy(&m.label, full + 19, 8);
        return m;
    }
};
