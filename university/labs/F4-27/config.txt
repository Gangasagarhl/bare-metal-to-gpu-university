# config.txt - F4-27: the firmware settings a Raspberry Pi 3 or 4 would need to start
# kernel8.img (k_d4.bin renamed) in 64-bit mode with the PL011 on GPIO 14/15.
# UNTESTED ON HARDWARE and UNVERIFIED: every option name below is recalled from memory and
# must be checked against the Raspberry Pi firmware documentation ("config.txt" chapter of
# the Raspberry Pi documentation; title only, not opened during this build) before use.
arm_64bit=1            # start the cores in AArch64
kernel=kernel8.img     # the file to load (at 0x80000 unless kernel_address says otherwise)
enable_uart=1          # turn on the primary UART at boot
dtoverlay=disable-bt   # Pi 3/4: give the PL011 back to GPIO 14/15 instead of Bluetooth
